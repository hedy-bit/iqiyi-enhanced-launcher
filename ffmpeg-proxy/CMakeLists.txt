cmake_minimum_required(VERSION 3.20)
project(IqiyiCodecObserver LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64")
endif()
add_library(codec_proxy SHARED proxy.cpp exports.def)
target_compile_features(codec_proxy PRIVATE cxx_std_17)
target_compile_definitions(codec_proxy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(codec_proxy PRIVATE /W4 /WX /utf-8)
set_property(TARGET codec_proxy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(codec_proxy PROPERTIES PREFIX "" OUTPUT_NAME "avcodec-61")
add_library(frame_tools SHARED frame_tools.cpp)
target_compile_features(frame_tools PRIVATE cxx_std_17)
target_compile_definitions(frame_tools PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(frame_tools PRIVATE /W4 /WX /utf-8)
set_property(TARGET frame_tools PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(frame_tools PROPERTIES PREFIX "" OUTPUT_NAME "iqiyi-frame-tools")
add_executable(test_frame_tools test_frame_tools.cpp)
target_compile_features(test_frame_tools PRIVATE cxx_std_17)
target_compile_definitions(test_frame_tools PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test_frame_tools PRIVATE frame_tools d3d11)
target_compile_options(test_frame_tools PRIVATE /W4 /WX /utf-8)
